$Texts = $Translations[$LangIndex]
if ($null -eq $Texts) {
    $Texts = $Translations[1] # Fallback to English
}

# 1. Wait for parent process to exit
if ($ParentPid -gt 0) {
    Write-Host "Waiting for parent process ($ParentPid) to exit..."
    while (Get-Process -Id $ParentPid -ErrorAction SilentlyContinue) {
        Start-Sleep -Seconds 1
    }
}

# 2. Check disk space
$DriveLetter = [System.IO.Path]::GetPathRoot($AppDir)
$Drive = Get-PSDrive -Name $DriveLetter[0] -ErrorAction SilentlyContinue
if ($null -eq $Drive) {
    $Drive = Get-PSDrive -Name "C"
}

$RequiredSpace = $AssetSize * 4 + 50MB
$FreeSpace = $Drive.Free

if ($FreeSpace -lt $RequiredSpace) {
    $ErrorMsg = $Texts.DiskSpaceError -f [Math]::Round($RequiredSpace / 1MB, 2), [Math]::Round($FreeSpace / 1MB, 2)
    [System.Windows.Forms.MessageBox]::Show(
        $ErrorMsg,
        $Texts.DiskSpaceErrorTitle,
        [System.Windows.Forms.MessageBoxButtons]::OK,
        [System.Windows.Forms.MessageBoxIcon]::Error
    )
    exit 1
}

# Configure TLS
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12 -bor [Net.SecurityProtocolType]::Tls13

$TempZip = Join-Path $env:TEMP "RealAmadeusUpdate.zip"
$TempExtractDir = Join-Path $env:TEMP "RealAmadeusExtract"

# Create UI components
$Form = New-Object System.Windows.Forms.Form
$Form.Text = $Texts.Title
$Form.Size = New-Object System.Drawing.Size(400, 160)
$Form.StartPosition = "CenterScreen"
$Form.FormBorderStyle = "FixedDialog"
$Form.MaximizeBox = $false
$Form.MinimizeBox = $false

$Label = New-Object System.Windows.Forms.Label
$Label.Location = New-Object System.Drawing.Point(20, 20)
$Label.Size = New-Object System.Drawing.Size(340, 25)
$Label.Text = $Texts.Preparing
$Label.Font = New-Object System.Drawing.Font("MS Gothic", 10)

$ProgressBar = New-Object System.Windows.Forms.ProgressBar
$ProgressBar.Location = New-Object System.Drawing.Point(20, 55)
$ProgressBar.Size = New-Object System.Drawing.Size(340, 25)
$ProgressBar.Style = "Continuous"
$ProgressBar.Minimum = 0
$ProgressBar.Maximum = 100
$ProgressBar.Value = 0

$Form.Controls.Add($Label)
$Form.Controls.Add($ProgressBar)

# Create WebClient and register events
$WebClient = New-Object System.Net.WebClient
$WebClient.Headers.Add("User-Agent", "RealAmadeusPC-Updater")

$global:DownloadCompleted = $false
$global:DownloadFailed = $false
$global:ErrorMsg = ""

$OnComplete = {
    if ($event.SourceEventArgs.Error -ne $null) {
        $global:DownloadFailed = $true
        $global:ErrorMsg = $event.SourceEventArgs.Error.Message
    }
    $global:DownloadCompleted = $true
}

Register-ObjectEvent -InputObject $WebClient -EventName DownloadFileCompleted -Action $OnComplete | Out-Null

# Trigger download on shown
$Form.Add_Shown({
    $WebClient.DownloadFileAsync((New-Object System.Uri($DownloadUrl)), $TempZip)
})

$Form.Show()

# Event loop for GUI during download
$LastProgress = -1
while (-not $global:DownloadCompleted) {
    $CurrentSize = 0
    if (Test-Path $TempZip) {
        $CurrentSize = (Get-Item $TempZip).Length
    }
    $Progress = 0
    if ($AssetSize -gt 0) {
        $Progress = [int]($CurrentSize / $AssetSize * 100)
    }
    if ($Progress -gt 100) { $Progress = 100 }

    if ($Progress -ne $LastProgress) {
        $LastProgress = $Progress
        $Label.Text = "$($Texts.Downloading) $Progress%"
        $ProgressBar.Value = $Progress
    }
    [System.Windows.Forms.Application]::DoEvents()
    Start-Sleep -Milliseconds 100
}

# Cleanup WebClient
Get-EventSubscriber | Unregister-Event
$WebClient.Dispose()

if ($global:DownloadFailed) {
    $ErrorMsg = $Texts.DownloadError -f $global:ErrorMsg
    [System.Windows.Forms.MessageBox]::Show(
        $ErrorMsg,
        $Texts.ErrorTitle,
        [System.Windows.Forms.MessageBoxButtons]::OK,
        [System.Windows.Forms.MessageBoxIcon]::Error
    )
    $Form.Close()
    exit 1
}

# 4. Extract update
$Label.Text = $Texts.Extracting
$ProgressBar.Style = "Marquee"
[System.Windows.Forms.Application]::DoEvents()

if (Test-Path $TempExtractDir) {
    Remove-Item -Path $TempExtractDir -Recurse -Force -ErrorAction SilentlyContinue
}
New-Item -ItemType Directory -Path $TempExtractDir -Force | Out-Null

try {
    Add-Type -AssemblyName System.IO.Compression.FileSystem
    [System.IO.Compression.ZipFile]::ExtractToDirectory($TempZip, $TempExtractDir)
} catch {
    $ErrorMsg = $Texts.ExtractError -f $_
    [System.Windows.Forms.MessageBox]::Show(
        $ErrorMsg,
        $Texts.ErrorTitle,
        [System.Windows.Forms.MessageBoxButtons]::OK,
        [System.Windows.Forms.MessageBoxIcon]::Error
    )
    Remove-Item -Path $TempZip -Force -ErrorAction SilentlyContinue
    $Form.Close()
    exit 1
}

# 5. Copy and overwrite files
$Label.Text = $Texts.Updating
[System.Windows.Forms.Application]::DoEvents()

$SourcePath = $TempExtractDir
$SubDirs = Get-ChildItem -Path $TempExtractDir
if ($SubDirs.Count -eq 1 -and $SubDirs[0].Attributes -match "Directory") {
    $SourcePath = $SubDirs[0].FullName
}

try {
    Copy-Item -Path "$SourcePath\*" -Destination $AppDir -Recurse -Force
} catch {
    $ErrorMsg = $Texts.UpdateError -f $_
    [System.Windows.Forms.MessageBox]::Show(
        $ErrorMsg,
        $Texts.ErrorTitle,
        [System.Windows.Forms.MessageBoxButtons]::OK,
        [System.Windows.Forms.MessageBoxIcon]::Error
    )
    Remove-Item -Path $TempZip -Force -ErrorAction SilentlyContinue
    Remove-Item -Path $TempExtractDir -Recurse -Force -ErrorAction SilentlyContinue
    $Form.Close()
    exit 1
}

# Cleanup
$Label.Text = $Texts.Cleaning
[System.Windows.Forms.Application]::DoEvents()
Remove-Item -Path $TempZip -Force -ErrorAction SilentlyContinue
Remove-Item -Path $TempExtractDir -Recurse -Force -ErrorAction SilentlyContinue

$Form.Close()

# 6. Completion Message and Restart
[System.Windows.Forms.MessageBox]::Show(
    $Texts.Completed,
    $Texts.CompletedTitle,
    [System.Windows.Forms.MessageBoxButtons]::OK,
    [System.Windows.Forms.MessageBoxIcon]::Information
)

Write-Host "Restarting application..."
$ExePath = Join-Path $AppDir $ExeName
if (Test-Path $ExePath) {
    Start-Process -FilePath $ExePath
} else {
    $NewExe = Get-ChildItem -Path $AppDir -Filter "*.exe" | Select-Object -First 1
    if ($null -ne $NewExe) {
        Start-Process -FilePath $NewExe.FullName
    }
}
